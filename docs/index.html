🚀 Quick Start

เริ่มใช้ใน 30 วินาที

ไม่ต้อง setup อะไรเลย — แค่ copy คำสั่ง แล้ววาง!

ลอง Health Check ก่อน

เช็คว่า API ออนไลน์อยู่ — ถ้าได้ {"status":"ok"} แสดงว่าพร้อมใช้งาน!

curl https://freeapi.kidpech.app/api/v1/health

สมัครสมาชิก

สร้าง account ใหม่ — จะได้ access token กลับมาใช้ทันที

curl -X POST https://freeapi.kidpech.app/api/v1/auth/register \
  -H "Content-Type: application/json" \
  -d '{
    "email": "you@example.com",
    "password": "MySecret123",
    "name": "Your Name"
  }'

Login & รับ Token

ใช้ email/password ที่สมัครไว้ — จะได้ access_token และ refresh_token

curl -X POST https://freeapi.kidpech.app/api/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email": "you@example.com", "password": "MySecret123"}'

สร้าง Profile ✨

ใช้ token ที่ได้มาเรียก protected endpoint — สร้างโปรไฟล์แรกของคุณ!

curl -X POST https://freeapi.kidpech.app/api/v1/profiles \
  -H "Authorization: Bearer YOUR_ACCESS_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "first_name": "สมชาย",
    "last_name": "ใจดี",
    "bio": "กำลังเรียนรู้ API 🚀"
  }'

  -H "Authorization: Bearer $TOKEN"

// Register

  method: 'POST',
  headers: { 'Content-Type': 'application/json' },

    email: 'test@example.com',
    password: 'MySecret123',
    name: 'Demo User'
  })

  method: 'POST',

  },
  body: JSON.stringify({
    first_name: 'Demo',
    last_name: 'User',
    bio: 'Made with JavaScript 🚀'
  })
}).then(r => r.json());

console.log(profile);

package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
)

const base = "https://freeapi.kidpech.app/api/v1"

func main() {
    // Login
    body, _ := json.Marshal(map[string]string{
        "email":    "test@example.com",
        "password": "MySecret123",
    })
    resp, _ := http.Post(base+"/auth/login",
        "application/json", bytes.NewBuffer(body))
    
    var result map[string]interface{}
    json.NewDecoder(resp.Body).Decode(&result)
    tokens := result["tokens"].(map[string]interface{})
    token := tokens["access_token"].(string)

    // Get Me
    req, _ := http.NewRequest("GET", base+"/users/me", nil)
    req.Header.Set("Authorization", "Bearer "+token)
    resp, _ = http.DefaultClient.Do(req)
    json.NewDecoder(resp.Body).Decode(&result)
    fmt.Println(result)
}
