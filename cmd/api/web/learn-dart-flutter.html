void main() {

  // Strings: Dart รองรับ Multiline และ Interpolation

  String bio = """

var vs final vs const — คนละความหมาย!

เรื่องนี้สำคัญมาก Senior Dev ต้องเลือกใช้ถูก:

// var — type ถูก lock หลังกำหนดค่าแรก
var name = "Dart"; // เป็น String ตลอดชีวิต
// name = 123; // ❌ Compile Error!

// final — กำหนดค่าครั้งเดียวได้ แต่รู้ค่าตอน runtime
final DateTime loginTime = DateTime.now(); // ทำได้ เพราะรู้ค่าตอนรัน

// const — ค่า Compile-time constant เท่านั้น (เร็วที่สุด!)
const double taxRate = 0.07;
const Color primaryColor = Color(0xFF00E5FF); // Flutter widget

// ⚡ Best Practice: ใช้ const เมื่อทำได้ เพราะ Flutter จะ cache Widget นั้น
// แทนที่จะสร้างใหม่ทุกครั้งที่ rebuild → ประหยัด Memory มาก

🛡️ Null Safety: หัวใจของ Dart ที่ปลอดภัย

ตั้งแต่ Dart 2.12 เป็นต้นมา Dart บังคับใช้ Sound Null Safety ซึ่งเป็นความแตกต่างใหญ่ที่สุดระหว่าง Dart กับภาษาอื่นๆ

ลองนึกภาพแบบนี้: ใน Java/Kotlin/Swift ตัวแปรเหมือนกล่องที่อาจ "ว่างเปล่า" (null) ได้ทุกใบ และเราไม่รู้ล่วงหน้า → ต้อง check ตลอด
แต่ใน Dart กล่องทุกใบ "ต้องมีของอยู่เสมอ" ยกเว้นกล่องที่มีป้าย ? ติดอยู่ ซึ่งหมายถึง "กล่องนี้อาจว่างได้" → Compiler บังคับให้ check ก่อนใช้

Non-nullable vs Nullable

// Non-nullable: ต้องมีค่าเสมอ (Default ใน Dart)
String username = "kidpech";
// username = null; ❌ Compile Error ทันที!

// Nullable: เพิ่ม ? เพื่อบอกว่า "อาจเป็น null ได้"
String? displayName; // null by default
displayName = "Kidpech App"; // ✅ ใส่ค่าได้
displayName = null;           // ✅ เป็น null ได้เช่นกัน

Null-aware Operators (อาวุธสำคัญ)

String? email;

// 1. ?. (Null-aware access) — เรียก method เฉพาะเมื่อไม่ null
int? length = email?.length; // null ถ้า email เป็น null

// 2. ?? (Null coalescing) — ค่า default ถ้า null
String display = email ?? "ยังไม่ได้ระบุ Email";

// 3. ??= (Null-aware assignment) — กำหนดค่าถ้ายังเป็น null อยู่
email ??= "default@example.com"; // set เฉพาะครั้งแรก

// 4. ! (Null assertion operator) — ⚠️ ใช้ด้วยความระวัง!
// บอก Dart ว่า "ฉันมั่นใจ 100% ว่าค่านี้ไม่ null"
String definitelyNotNull = email!; // Runtime Error ถ้า null จริง!

// Real-world Pattern: Early Return (Clean & Safe)
Future<void> updateProfile(String? name) async {
  if (name == null || name.isEmpty) return; // Guard clause
  await api.update(name); // ถึงบรรทัดนี้ name ไม่ null แน่นอน
}

📚 Collections & Functional Programming

List, Map, Set คือหัวใจ แต่บรรดา Senior Dev จะเขียนแบบ Functional ซึ่งกระชับและ Bug-free กว่ามาก

List: มากกว่าแค่ Array

// การสร้าง List แบบต่างๆ
final primes = [2, 3, 5, 7, 11]; // Type inferred เป็น List<int>
final fixed = List<String>.filled(5, ""); // 5 ช่อง เริ่มต้นด้วย ""
final generated = List.generate(10, (i) => i * i); // [0,1,4,9,16,...]

// Spread operator (...) — รวม List ง่ายๆ
final extra = [13, 17];
final allPrimes = [...primes, ...extra]; // [2,3,5,7,11,13,17]

// Collection if/for — สร้าง List แบบมีเงื่อนไข
bool includeAdmin = true;
final menu = [
  'หน้าหลัก',
  'โปรไฟล์',
  if (includeAdmin) 'จัดการระบบ', // เพิ่มเฉพาะเมื่อ true
];

Functional Style: map, where, reduce, fold

อย่าเขียน for loop บ้านๆ ให้ใช้ functional style ที่อ่านง่ายกว่า

final products = [
  Product(name: 'iPhone', price: 35000, inStock: true),
  Product(name: 'iPad', price: 25000, inStock: false),
  Product(name: 'MacBook', price: 55000, inStock: true),
];

// where: filter
final inStock = products.where((p) => p.inStock);

// map: transform
final prices = products.map((p) => p.price); // [35000, 25000, 55000]

// fold: aggregate (เหมือน reduce แต่กำหนด initial value ได้)
final totalValue = products
    .where((p) => p.inStock)
    .fold(0.0, (sum, p) => sum + p.price); // 90000.0

// Method chaining (Chain ได้เลย สวยกว่า Loop มาก)
final summary = products
    .where((p) => p.inStock)
    .map((p) => '\${p.name}: ฿\${p.price}')
    .join(', '); // "iPhone: ฿35000, MacBook: ฿55000"

📦 Modern Variables, Records & Patterns

ลืมการประกาศตัวแปรแบบเดิมๆ มาดูวิธีที่ทันสมัยและ Clean กว่ากันครับ

1. Records (New in Dart 3)

ไม่ต้องสร้าง Class เล็กๆ น้อยๆ เพื่อ return ค่าหลายค่าอีกต่อไป

// เก่า: ต้องสร้าง class LatLng
// ใหม่: ใช้ Records!
(double, double) getLocation(String city) {
  if (city == 'Bangkok') return (13.75, 100.50);
  return (0.0, 0.0);
}

void main() {
  // Destructuring (แกะค่าออกมาทันที)
  var (lat, lng) = getLocation('Bangkok'); 
  print('Lat: $lat, Lng: $lng');
  
  // Named Records (ระบุชื่อได้ด้วย)
  ({String name, int age}) user = (name: 'Somchai', age: 25);
  print(user.name);
}

2. Pattern Matching (Switch Expression)

การเขียน Switch แบบเดิมมันยาวเกะกะ มาดูแบบ Expression สั้นๆ สวยๆ

String getStatusMessage(String status) => switch (status) {
  'PROCESSED' => '✅ ดำเนินการเสร็จสิ้น',
  'PENDING'   => '⏳ กำลังรอ...',
  'FAILED' || 'ERROR' => '❌ เกิดข้อผิดพลาด', // Match หลายเงื่อนไขได้
  _           => '❓ ไม่ทราบสถานะ',           // Default case
};

// ใช้กับ Records ก็ได้นะ!
String authGuard((String, bool) context) => switch (context) {
  ('admin', true)  => 'Allow Admin',
  ('user', true)   => 'Allow User',
  (_, false)       => 'Redirect to Login',
  _                => 'Access Denied',
};

🔨 Functions & Null Safety ขั้นสูง

Required Named Parameters + Default Value

ท่าไม้ตายของการออกแบบ API ใน Dart ให้ใช้งานง่าย

// การออกแบบ Config ที่ดี
class ServerConfig {
  final String host;
  final int port;
  final bool isSecure;

  // Constructor แบบนี้เรียกใช้ง่ายสุดๆ
  ServerConfig({
    required this.host, 
    this.port = 8080,    // Default value
    this.isSecure = false,
  });
}

void main() {
  // คนใช้ไม่ต้องจำลำดับ parameter และรู้ว่าอะไรจำเป็นต้องใส่
  final config = ServerConfig(
    host: 'api.example.com',
    isSecure: true, // port 8080 auto
  );
}

Generics: เขียนครั้งเดียวใช้ได้ครอบจักรวาล

หัวใจสำคัญของการเขียน Library หรือ Utility Class ให้ Reusable

// ไม่มี Generics: ต้องเขียน class แยกสำหรับ Integer, String, User... เหนื่อยแย่
// มี Generics (T): เขียนทีเดียวจบ
class ApiResponse<T> {
  final bool success;
  final T? data;        // data เป็น type อะไรก็ได้
  final String? error;

  ApiResponse.success(this.data) : success = true, error = null;
  ApiResponse.failure(this.error) : success = false, data = null;
}

void main() {
  // Type Safety ยังอยู่ครบ!
  final userResponse = ApiResponse<User>.success(User(name: "Somchai"));
  final productResponse = ApiResponse<List<Product>>.success([p1, p2]);
}

Extension Methods: เพิ่มพลังให้ Class เดิม

ไม่ต้องสร้าง Class ใหม่ (Utils) ให้รก Extension ช่วยให้เราเติม method ลงไปใน class ที่มีอยู่แล้วได้ (แม้แต่ class ของ Dart เอง)

extension StringValidators on String {
  bool get isValidEmail {
    return RegExp(r'^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$').hasMatch(this);
  }
}

extension ContextTheme on BuildContext {
  // เรียกง่ายกว่า Theme.of(context).textTheme.headlineLarge
  TextStyle? get headlineLarge => Theme.of(this).textTheme.headlineLarge;
  Color get primaryColor => Theme.of(this).colorScheme.primary;
}

// การใช้งาน
if (email.isValidEmail) { ... } // อ่านง่ายเหมือนภาษาพูด
Text("Hello", style: context.headlineLarge); // สั้นกระชับ

🏗️ Class Modifiers (ของใหม่ต้องรู้!)

ใน Dart 3.0 เราควบคุมการสืบทอดได้ละเอียดมาก เพื่อป้องกันการใช้งานผิดๆ ในทีมขนาดใหญ่

Sealed Class (พระเอกของ State Management)

sealed บังคับให้เรารู้จัก subclass ทั้งหมด เหมาะมากสำหรับทำ State ของหน้าจอ

// ประกาศ State ที่เป็นไปได้ทั้งหมด
sealed class ApiState {}

class Loading extends ApiState {}
class Success extends ApiState {     
  final String data;
  Success(this.data);
}
class Error extends ApiState {
  final String message;
  Error(this.message);
}

// เวลาเอาไปใช้ Dart จะบังคับให้เรา handle ให้ครบทุก case (Exhaustive Check)
// ถ้าลืม case ไหนไป จะ Compile Error ทันที! ปลอดภัยมาก
String buildUI(ApiState state) => switch (state) {
  Loading() => 'Spinner...',
  Success(data: var d) => 'Got data: $d', // Destructure ค่าออกมาใช้ได้เลย
  Error(message: var m) => 'Alert: $m',
};

Interface vs Base vs Final

  • interface class: บังคับให้คนอื่น implement เท่านั้น ห้าม extend (เหมาะสำหรับทำสัญญา API)
  • base class: บังคับให้คนอื่น extend เท่านั้น ห้าม implement (เหมาะสำหรับบังคับ implementation logic)
  • final class: ปิดตาย! ห้ามสืบทอด ห้าม implement นอก library

🔧 Constructors: ทุกรูปแบบที่ต้องรู้

Dart มี Constructor หลายแบบมากกว่าภาษาอื่น ซึ่งเป็น Feature ที่ทรงพลังมากสำหรับการออกแบบ API ที่ดี

5 แบบ Constructor ที่ใช้บ่อยมาก

class User {
  final String id;
  final String name;
  final String? email;
  final DateTime createdAt;

  // 1. Default Constructor (แบบย่อ — Syntactic Sugar)
  User({required this.id, required this.name, this.email})
      : createdAt = DateTime.now(); // Initializer list

  // 2. Named Constructor — สร้างจาก context ต่างๆ
  User.fromJson(Map<String, dynamic> json)
      : id = json['id'],
        name = json['name'],
        email = json['email'],
        createdAt = DateTime.parse(json['created_at']);

  // 3. Factory Constructor — ควบคุม Object สร้างได้ 100%
  //    (Return existing instance, Singleton, หรือ subtype ได้!)
  factory User.guest() {
    return User(id: 'guest-0', name: 'Guest User');
  }

  // 4. Const Constructor — Object ถูก Cache ไว้ (เร็วมากใน Flutter)
  //    ใช้ได้เฉพาะเมื่อ field ทุกตัวเป็น final
  // const User({...}); ← ทำได้ถ้า createdAt ไม่อยู่

  // 5. Redirect Constructor
  User.admin({required String name}) :
    this(id: 'admin-${name.hashCode}', name: name, email: 'admin@example.com');
}

// การใช้งาน
final u1 = User(id: 'u1', name: 'Kidpech');
final u2 = User.fromJson({'id': 'u2', 'name': 'Dev', 'email': null, 'created_at': '2026-01-01T00:00:00Z'});
final guest = User.guest();
final admin = User.admin(name: 'SuperAdmin');

🧬 Inheritance, Abstract & Mixins

การสืบทอดใน Dart มี 3 เครื่องมือหลักที่ใช้งานต่างกัน: extends, implements, และ with

Abstract Class: สัญญาที่มี Default Implementation

// Abstract Class เหมาะสำหรับ "ฐาน" ที่มี Logic บางส่วน
abstract class Repository<T> {
  Future<T?> findById(String id);
  Future<List<T>> findAll();
  Future<void> save(T entity);
  Future<void> delete(String id);

  // Default implementation — subclass ใช้ได้เลย
  Future<bool> exists(String id) async {
    return (await findById(id)) != null;
  }
}

class UserRepository extends Repository<User> {
  @override
  Future<User?> findById(String id) async {
    return await database.query('SELECT * FROM users WHERE id = ?', [id]);
  }
  // ต้อง implement ทุก abstract method
  @override Future<List<User>> findAll() async => [...];
  @override Future<void> save(User u) async => [...];
  @override Future<void> delete(String id) async => [...];
}

Mixins: "พลังพิเศษที่ถอด-ใส่ได้"

Mixin ไม่ใช่ Class ที่สืบทอดได้ แต่คือ "ชุดความสามารถ" ที่นำไปติดกับ Class ใดก็ได้

// Real-world Example: Flutter Animation มักใช้ Mixin แบบนี้
mixin LoggableMixin {
  String get logTag => runtimeType.toString();

  void log(String message) {
    print('[\$logTag] \$message');
  }
}

mixin RetryableMixin {
  final int maxRetries = 3;

  Future<T> withRetry<T>(Future<T> Function() fn) async {
    for (int i = 0; i < maxRetries; i++) {
      try {
        return await fn();
      } catch (e) {
        if (i == maxRetries - 1) rethrow;
        await Future.delayed(Duration(seconds: 1 << i)); // Exponential backoff
      }
    }
    throw StateError('Should not reach here');
  }
}

// รวมพลัง!
class ProductService with LoggableMixin, RetryableMixin {
  Future<List<Product>> fetchProducts() async {
    log('Fetching products...');
    return withRetry(() => api.getProducts()); // retry อัตโนมัติสูงสุด 3 ครั้ง
  }
}

class QueryBuilder {
  String _table = '';
  final List<String> _conditions = [];
  int? _limit;
  String? _orderBy;

  QueryBuilder from(String table) { _table = table; return this; }
  QueryBuilder where(String condition) { _conditions.add(condition); return this; }
  QueryBuilder limit(int n) { _limit = n; return this; }
  QueryBuilder orderBy(String field) { _orderBy = field; return this; }

  String build() {
    var query = 'SELECT * FROM \$_table';
    if (_conditions.isNotEmpty) query += ' WHERE \${_conditions.join(' AND ')}';
    if (_orderBy != null) query += ' ORDER BY \$_orderBy';
    if (_limit != null) query += ' LIMIT \$_limit';
    return query;
  }
}

⚡ Isolates: True Multithreading ใน Dart

Dart เป็น Single-threaded แต่ใช้ Event Loop เหมือน JavaScript ทีนี้ถ้า Task หนักมาก (parse JSON ล้านบรรทัด, compress image) มันจะ Block UI Thread ทำให้หน้าค้าง (Jank)!

คำตอบคือ Isolates — กระบวนการ Dart ที่ทำงานแยกอิสระพร้อมกัน โดย ไม่แชร์ Memory กัน (คุยกันผ่าน Message เท่านั้น)

Analogy: Isolate คืออะไร?

คิดว่า Isolate คือ "คนงานคนละคน" ที่อยู่คนละห้อง ห้ามยืมของกัน ต้องส่งโน้ตผ่านใต้ประตูเท่านั้น (Message Passing). ต่างกับ Thread ในภาษาอื่นที่แชร์ Memory ได้เลยแต่เสี่ยง Race Condition.

วิธีที่ 1: compute() — ง่ายที่สุด (Flutter Only)

เหมาะเมื่อ "ส่งข้อมูลไป ประมวลผล คืนค่ากลับมา" แบบง่ายๆ

// ฟังก์ชั่นที่จะรันใน Isolate ต้องเป็น top-level function เท่านั้น!
List<Product> _parseProductsInBackground(String jsonStr) {
  final List decoded = jsonDecode(jsonStr); // Heavy computation
  return decoded.map((j) => Product.fromJson(j)).toList();
}

// เรียกใช้จาก UI Thread — ไม่ค้างหน้าแม้แต่ frame เดียว
final products = await compute(_parseProductsInBackground, rawJsonString);
print('Got \${products.length} products');

วิธีที่ 2: Isolate.run() — Dart 2.19+

// Dart 2.19+ มี Isolate.run() ที่ใช้งานง่ายกว่ามาก
final result = await Isolate.run(() async {
  // ทำของหนักที่นี่ (รองรับ closure แต่ capture ได้เฉพาะ data ที่ copy ได้)
  return await _expensiveEncryption(largeData);
});

ข้อควรระวัง (Senior Dev ต้องรู้!)

  • Isolate ใหม่ต้องโหลด Dart VM ใหม่ — มี overhead สูง อย่าสร้าง-ทำลายบ่อย
  • ใช้ Isolate Pool pattern หรือ IsolateNameServer สำหรับงานต่อเนื่อง
  • Platform Channels (Flutter ↔ Native) ต้องเรียกจาก Main Isolate เท่านั้น
  • Object ที่ส่งข้ามไปต้องเป็น primitive, List, Map หรือ implement SendPort เท่านั้น

🌳 Widget Tree: 3 Trees ที่ Flutter ใช้จริง

Flutter ไม่ได้มีแค่ Widget Tree เดียว แต่มี 3 Trees ทำงานร่วมกัน การเข้าใจเรื่องนี้ทำให้ Debug Performance ได้เฉียบคมขึ้นมาก

3 Trees ของ Flutter

  Widget Tree           Element Tree          Render Tree
  (Blueprint)           (Instance)            (Actual Pixels)
  ─────────────         ─────────────         ─────────────
  Column()        →     ColumnElement   →     RenderFlex
    Text("Hi")    →     TextElement     →     RenderParagraph
    Container()   →     ContainerElem   →     RenderConstrainedBox
  • Widget Tree: แค่ "คำอธิบาย" (immutable config) สร้าง-ทำลายได้ถูกมาก
  • Element Tree: "ชีวิตจริง" ของ Widget เก็บ State, parent-child relationship (ทนทาน)
  • Render Tree: คำนวณ Layout, Paint ลงจอ (หนักสุด)
Widget = แบบสถาปัตย์ | Element = ตัวอาคารจริง | RenderObject = งานก่อสร้าง
การ rebuild Widget ไม่ได้ทำลาย Element เสมอไป — Flutter ฉลาดพอที่จะ reuse Element ถ้า Widget type เหมือนกัน

Key: เบาะแสให้ Flutter ระบุ Element

// ปัญหา: swap เด็ก 2 คนโดยไม่ใช้ Key → Flutter สับสน, State หาย!
Row(children: [ColorBox(color: red), ColorBox(color: blue)])
// เปลี่ยน order:
Row(children: [ColorBox(color: blue), ColorBox(color: red)])  // State ยังผิดอยู่!

// แก้ไข: ใช้ Key บอก Flutter ว่า Widget ไหนคือใคร
Row(children: [
  ColorBox(key: ValueKey('red'), color: red),
  ColorBox(key: ValueKey('blue'), color: blue),
]) // ✅ State ถูกต้องแม้ swap order

// GlobalKey: ใช้เข้าถึง State ของ Widget จากภายนอก (ใช้น้อยที่สุดเพราะ couple code)
final formKey = GlobalKey<FormState>();
formKey.currentState?.validate();

🔄 Stateless vs Stateful Widget: Lifecycle ที่ต้องรู้ใจ

StatelessWidget: ไม่มีความจำ (Functional UI)

ทุกครั้งที่ Parent rebuild มัน recreate Widget ใหม่ทั้งหมด เหมาะสำหรับ UI ที่ขึ้นอยู่กับ Input เท่านั้น

// ดีที่สุดสำหรับ Presentational Widgets
class ProductCard extends StatelessWidget {
  final Product product;
  final VoidCallback onTap;

  const ProductCard({super.key, required this.product, required this.onTap});

  @override
  Widget build(BuildContext context) {
    // ทุก build จะ recreate ทั้งหมด (แต่ไม่แพงเพราะ Widget เป็นแค่ config)
    return GestureDetector(
      onTap: onTap,
      child: Card(child: Text(product.name)),
    );
  }
}

StatefulWidget: มีความจำ (Mutable State)

Widget + State Object แยกกัน Flutter เก็บ State ไว้ใน Element Tree แม้ Widget ถูก rebuild

class CounterWidget extends StatefulWidget {
  const CounterWidget({super.key});

  @override
  State<CounterWidget> createState() => _CounterWidgetState();
}

class _CounterWidgetState extends State<CounterWidget>
    with SingleTickerProviderStateMixin {
  int _count = 0;
  late AnimationController _animCtrl;

  // ── LIFECYCLE ──────────────────────────────────────────

  @override
  void initState() {
    super.initState();
    // เรียกครั้งเดียวตอน Widget ถูก insert เข้า Tree
    // เหมาะสำหรับ: init controller, subscribe stream, fetch data ครั้งแรก
    _animCtrl = AnimationController(vsync: this, duration: 300.ms);
    _fetchInitialData();
  }

  @override
  void didUpdateWidget(CounterWidget oldWidget) {
    super.didUpdateWidget(oldWidget);
    // เรียกทุกครั้งที่ Parent ส่ง config ใหม่มา (Widget rebuilt แต่ State ยังอยู่)
    // เหมาะสำหรับ: update local state เมื่อ props เปลี่ยน
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    // เรียกหลัง initState และเมื่อ InheritedWidget ที่ depend เปลี่ยน
    // เหมาะสำหรับ: ดึงค่าจาก Theme, MediaQuery, Provider
    final theme = Theme.of(context); // Safe to call here!
  }

  @override
  void dispose() {
    // เรียกตอน Widget ถูก remove ออกจาก Tree — ต้อง cleanup เสมอ!
    _animCtrl.dispose();       // ❗ ไม่ dispose → memory leak!
    _subscription?.cancel();  // ❗ ไม่ cancel → zombie listener!
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    // เรียกบ่อยที่สุด — อย่าทำงานหนักที่นี่!
    return Text('\$_count');
  }
}

Rule of Thumb: เลือกอะไรดี?

  • ไม่มี local state, ไม่มี animation, ไม่ subscribe อะไร → StatelessWidget
  • มี Counter/Toggle/Form/Animation → StatefulWidget
  • State แชร์ข้าม Widget หลายตัว → Riverpod / BLoC (อย่าเก็บไว้ใน StatefulWidget)

🏆 Riverpod: The Modern Standard

Riverpod คือร่างทองของ Provider ที่แก้ปัญหาทุกอย่าง และให้เราจัดการ Async State (Loading/Error/Data) ได้ง่ายที่สุดในโลก

Riverpod ไม่ใช่แค่ State Management แต่มันคือ "Reactive Caching Framework" (ช่วยจำค่า, แชร์ค่า, และรีเฟรชค่าเมื่อจำเป็น) 
// 1. สร้าง Provider แบบ Async (FutureProvider) เหมาะกับการดึง API
final productProvider = FutureProvider.family<List<Product>, String>((ref, category) async {
  final api = ref.watch(apiClientProvider);
  return api.fetchProducts(category); // คืนค่า Future
});

// 2. ใช้ใน Widget (Clean สุดๆ ด้วย AsyncValue)
class ProductPage extends ConsumerWidget {
  @override
  Widget build(BuildContext context, WidgetRef ref) {
    // Watch: เมื่อค่าเปลี่ยน Widget จะ rebuild เฉพาะตรงนี้
    final asyncProducts = ref.watch(productProvider('shoes'));
    
    // Pattern Matching กับ UI 3 สถานะ (Loading/Error/Data) อัตโนมัติ!
    return asyncProducts.when(
      data: (products) => ListView.builder(itemCount: products.length, itemBuilder: ...),
      error: (err, stack) => ErrorWidget(err),
      loading: () => Center(child: CircularProgressIndicator()),
    );
  }
}

ทำไมโค้ดนี้ถึง "เทพ"?

  • ไม่ต้องเขียน `if (isLoading)` เอง Riverpod จัดการให้
  • ถ้า user ออกจากหน้า `ProductPage` และกลับเข้ามาใหม่ Riverpod จะใช้ **Cache** เดิมถ้ายังไม่หมดอายุ (ไม่ต้องโหลดใหม่)
  • ถ้าเกิด Error มันจะเก็บ state error ไว้ และเราสามารถสั่ง `ref.refresh(productProvider)` เพื่อลองใหม่ได้ง่ายๆ

BLoC (Business Logic Component)

ยังคงเป็น King 👑 ในฝั่ง Enterprise (Banking/FinTech) และทีมขนาดใหญ่ เพราะ:

  • Event-Driven: บังคับโครงสร้าง Event -> State ชัดเจน (Strict)
  • Traceability: รู้หมดว่า State เปลี่ยนเพราะ Event อะไร (ผ่าน BlocObserver)
  • Testability: เทส logic ง่ายมาก เพราะ input คือ Event และ output คือ State เสมอ

BLoC ตัวอย่างจริง: Auth Flow

// 1. Events — สิ่งที่ user หรือระบบทำ (immutable!)
sealed class AuthEvent {}
final class LoginRequested extends AuthEvent {
  final String email, password;
  LoginRequested(this.email, this.password);
}
final class LogoutRequested extends AuthEvent {}
final class SessionRefreshRequested extends AuthEvent {}

// 2. States — สิ่งที่ UI จะแสดง (immutable!)
sealed class AuthState {}
final class AuthInitial extends AuthState {}
final class AuthLoading extends AuthState {}
final class AuthAuthenticated extends AuthState {
  final User user;
  AuthAuthenticated(this.user);
}
final class AuthError extends AuthState {
  final String message;
  AuthError(this.message);
}

// 3. BLoC — Business Logic อยู่ที่นี่ทั้งหมด
class AuthBloc extends Bloc<AuthEvent, AuthState> {
  final AuthRepository _authRepository;

  AuthBloc(this._authRepository) : super(AuthInitial()) {
    on<LoginRequested>(_onLoginRequested);
    on<LogoutRequested>(_onLogoutRequested);
  }

  Future<void> _onLoginRequested(
    LoginRequested event,
    Emitter<AuthState> emit,
  ) async {
    emit(AuthLoading());
    try {
      final user = await _authRepository.login(event.email, event.password);
      emit(AuthAuthenticated(user));
    } catch (e) {
      emit(AuthError(e.toString()));
    }
  }

  Future<void> _onLogoutRequested(LogoutRequested event, Emitter<AuthState> emit) async {
    await _authRepository.logout();
    emit(AuthInitial());
  }
}

// 4. BlocObserver — Monitor ทุกอย่าง! (Logging, Analytics)
class AppBlocObserver extends BlocObserver {
  @override
  void onEvent(Bloc bloc, Object? event) {
    super.onEvent(bloc, event);
    print('[\${bloc.runtimeType}] Event: \$event'); // Log ทุก Event
  }
  @override
  void onError(BlocBase bloc, Object error, StackTrace stackTrace) {
    Crashlytics.record(error, stackTrace); // Crash report อัตโนมัติ
    super.onError(bloc, error, stackTrace);
  }
}

// 5. ใช้ใน Widget (Pattern Matching กับ sealed class!)
BlocConsumer<AuthBloc, AuthState>(
  listener: (ctx, state) {
    if (state is AuthAuthenticated) Navigator.pushNamed(ctx, '/home');
  },
  builder: (ctx, state) => switch (state) {
    AuthLoading()         => CircularProgressIndicator(),
    AuthAuthenticated(user: final u) => Text('Welcome, \${u.name}'),
    AuthError(message: final msg)    => Text('Error: \$msg', style: TextStyle(color: Colors.red)),
    _                     => LoginForm(),
  },
)

🌐 Networking: อย่าใช้ http package ตรงๆ!

ในงาน Enterprise เราต้องจัดการ Token Refresh, Error Handling, และ Timeout แบบรวมศูนย์ แนะนำให้ใช้ Dio เสมอ

Why Dio? (The Supercharged HTTP Client)

Dio มีฟีเจอร์ "Interceptors" ที่เหมือน Middleware ใน Backend ช่วยดัดแปลง request/response ได้ก่อนถึงมือเรา

final dio = Dio(BaseOptions(baseUrl: 'https://api.example.com'));

void setupInterceptors() {
  dio.interceptors.add(InterceptorsWrapper(
    onRequest: (options, handler) async {
      // 1. แปะ Token อัตโนมัติทุก request
      final token = await secureStorage.read(key: 'jwt_token');
      options.headers['Authorization'] = 'Bearer $token'; 
      return handler.next(options);
    },
    onError: (DioException e, handler) async {
      // 2. ถ้าเจอ 401 Unauthorized -> ลอง Refresh Token ก่อน
      if (e.response?.statusCode == 401) {
         if (await refreshToken()) {
            return handler.resolve(await dio.fetch(e.requestOptions)); // Retry
         }
      }
      return handler.next(e);
    },
  ));
}

โครงสร้าง Folder ที่แนะนำสำหรับโปรเจกต์ขนาดกลาง-ใหญ่ (Feature-first Structure):

lib/
├── core/                  # ของกลางที่ใช้ร่วมกัน (Error, Network, DI)
├── features/              # แยกตาม Feature (แนะนำกว่าแยกตาม Layer)
│   ├── auth/
│   │   ├── data/          # 🧱 Data Layer (คุยกับโลกภายนอก)
│   │   │   ├── datasources/ # Remote (API), Local (Storage)
│   │   │   ├── models/      # JSON DTOs (Data Transfer Objects)
│   │   │   └── repositories/# Logic การเลือก source (API หรือ Cache?)
│   │   ├── domain/        # 🧠 Domain Layer (สมองของธุรกิจ)
│   │   │   ├── entities/    # Plain Dart Class (ห้ามมี JSON logic!)
│   │   │   ├── repositories/# Abstract Interface (สัญญาว่าทำอะไรได้)
│   │   │   └── usecases/    # Business Logic จริงๆ (Login, Validate)
│   │   └── presentation/  # 🖼️ Presentation Layer (หน้าจอ)
│   │       ├── bloc/        # State Management (BloC/Riverpod)
│   │       ├── pages/       # Screens
│   │       └── widgets/     # Local Widgets
│   └── product/
└── main.dart

🚫 ข้อห้ามสำคัญของ Clean Arch

  • ห้าม Presentation รู้จัก Data Layer: UI ต้องเรียกผ่าน Domain/Usecase เท่านั้น ห้ามเรียก API ตรงๆ
  • ห้าม Domain รู้จัก Flutter: Domain Layer ต้องเป็น Pure Dart (ไม่มี `import 'package:flutter/...'`) เพื่อให้ Test ง่ายและ Portable
  • Data Layer เป็นทาส: มีหน้าที่แค่หาข้อมูลมาให้ Domain ไม่ต้องคิด Logic ซับซ้อน

🔄 Data Modeling: code-gen คือเพื่อนรัก

อย่าเขียน fromJson / toJson เอง! ผิดง่ายและเสียเวลา

Freezed + JsonSerializable

แพ็คเกจคู่บุญที่ต้องมีทุกโปรเจกต์ freezed ช่วยทำ Immutable Data Class + CopyWith + Equality Check ให้ฟรีๆ

@freezed
class User with _$User {
  // สร้าง Constructor และระบุ field
  const factory User({
    required int id,
    required String username,
    @Default(false) bool isActive, // กำหนด default value
  }) = _User;

  // บอกให้สร้าง fromJson ให้ด้วย
  factory User.fromJson(Map<String, dynamic> json) => _$UserFromJson(json);
}

🧪 Automated Testing: เขียนโค้ดให้หลับสบาย

ใน Enterprise, "โค้ดที่ไม่มี Test = Legacy Code". เรามี Test 3 ระดับที่ต้องรู้จัก:

1. Unit Test (ทดสอบ Logic) FASTEST

ทดสอบ Usecase, Repositories, หรือ Util Functions ว่าทำงานถูกต้องตาม Business Logic ไหม

test('User should be valid if email is correct', () {
  final user = User(email: 'test@example.com');
  expect(user.isValid, true);
});

// ใช้ Mocktail เพื่อจำลอง dependencies (Network/DB)
when(() => mockApi.login()).thenAnswer((_) async => 'token123');

2. Widget Test (ทดสอบ UI) MEDIUM

ทดสอบ Component ว่าแสดง widget ถูกต้องไหมเมื่อ state เปลี่ยนแปลง โดยไม่ต้องรันแอปจริง

testWidgets('Counter increments smoke test', (WidgetTester tester) async {
  await tester.pumpWidget(MyApp());
  
  expect(find.text('0'), findsOneWidget); // เริ่มต้นที่ 0
  await tester.tap(find.byIcon(Icons.add)); // กดปุ่ม +
  await tester.pump(); // รอรีเฟรชหน้าจอ
  
  expect(find.text('1'), findsOneWidget); // ต้องเป็น 1
});

3. Golden Test (ทดสอบ Pixel Perfect) SLOWEST

ถ่ายรูปหน้าจอเทียบกับรูปต้นฉบับ (Master) ถ้ารูปต่างกันแม้แต่ 1 pixel -> Test Failed! (เหมาะสำหรับกัน UI เพี้ยนโดยไม่ตั้งใจ)

🚀 Performance Tuning: Secrets of 60 FPS

ทำแอปให้ใช้ได้ กับทำแอปให้ "ลื่นปรื๊ด" มันต่างกันตรงความเข้าใจเรื่อง Widget Tree

Common Mistakes (ข้อผิดพลาดที่เจอบ่อย)

  • SizedBox vs Container: ถ้าแค่ต้องการเว้นระยะห่าง ใช้ `SizedBox` เสมอ (เร็วกว่า `Container` มาก เพราะไม่มี painting logic)
  • const keywords: ใส่ `const` หน้า Widget ทุกครั้งที่ทำได้ (ลดการสร้าง object ใหม่ memory churn)
  • Transparency: อย่าใช้ Opacity widget โดยไม่จำเป็น เพราะมันบังคับให้ Flutter วาดลง buffer (แพงมาก) ให้ใช้ `Color.withOpacity()` แทนถ้าทำได้

ListView.builder & Image Cache

  • อย่าใช้ `ListView(children: ...)` ถ้ามีของเกิน 20 ชิ้น ให้ใช้ `.builder` เพื่อ render เฉพาะสิ่งที่เห็น (Lazy Loading)
  • ใช้ `cached_network_image` เสมอ ห้ามโหลดรูปเดิมซ้ำๆ จากเน็ต
  • ถ้า Animation กระตุก ลองครอบด้วย `RepaintBoundary` (แยก layer การวาด ไม่กระทบส่วนอื่น)

🌟 Enterprise Must-Haves (สิ่งที่ขาดไม่ได้ในปี 2026)

1. Internationalization (i18n)

❌ ห้าม Hardcode String เด็ดขาด! (เช่น `Text("สวัสดี")`)

✅ ใช้ Library เช่น slang หรือ easy_localization เสมอ แม้จะทำแอปภาษาเดียว (เผื่ออนาคต)

Text(context.t.welcome_message(name: user.name)) // Type-safe translations!

2. Dependency Injection (DI)

เลิกส่ง Service ต่อๆ กันเป็นทอดผ่าน Constructor (Prop Drilling). ใช้ get_it หรือ Riverpod เพื่อเข้าถึง Dependencies จากที่ไหนก็ได้

// BAD: PageA -> PageB -> PageC (ส่ง apiService ต่อกันไปเรื่อยๆ)
// GOOD: Ref.read(apiProvider) (เรียกใช้ได้เลยเมื่อต้องการ)

3. Flavor & Environment

แอปจริงต้องมี Dev, Staging, Production แยกกัน (Database คนละก้อน) ใช้ --dart-define หรือ package envied เพื่อจัดการ Secrets อย่างปลอดภัย

Senior Developer Checklist:

  • Async Gap: ระวังการใช้ `context` หลัง `await` (เพราะหน้าจออาจถูกปิดไปแล้ว) -> เช็ค `if (context.mounted)` เสมอ
  • Don't fight the framework: อย่าพยายามฝืน Flutter โดยการเขียน Native UI ผสมเยอะเกินไป, ให้เชื่อใจ Skia/Impeller Engine
  • Responsive: ออกแบบเผื่อจอพับ / Tablet เสมอ (ใช้ `LayoutBuilder`)
