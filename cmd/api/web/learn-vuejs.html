Vue.js คืออะไร? (แบบเด็ก 5 ขวบก็เข้าใจ)

สมมติว่าเว็บไซต์คือ "หนังสือ" — HTML เป็นเนื้อหา, CSS เป็นการตกแต่ง, JavaScript เป็นเวทมนตร์ที่ทำให้หนังสือขยับได้

Vue.js คือ "ไม้กายสิทธิ์" ที่ช่วยให้คุณร่ายเวทมนตร์ JavaScript ได้ง่ายขึ้น 10 เท่า! แทนที่จะเขียน document.getElementById เอง Vue จัดการให้หมด

ทำไมต้อง Vue.js ในปี 2026?

  • เรียนรู้ง่ายที่สุด ในบรรดา 3 Framework ใหญ่ (React, Vue, Angular) — ซึ่งไม่ได้แปลว่ามันอ่อน แต่มัน "มีความเป็นมิตร" กับคนเริ่มต้น
  • Composition API — ระบบใหม่ที่ทำให้จัดระเบียบโค้ดได้สะอาดกว่าเดิมมาก (คล้าย React Hooks แต่ง่ายกว่า)
  • TypeScript First — Vue 3 เขียนด้วย TypeScript ทั้ง Runtime ทำให้ Type Safety ดีเลิศ
  • Vite — Dev Server เร็วที่สุดในโลก (สร้างโดย Evan You คนเดียวกับที่สร้าง Vue)
  • Nuxt 3 — Full-stack Framework ที่เหมือน Next.js ของ React แต่เป็น Vue version
  • Ecosystem สมบูรณ์ — Pinia (State), Vue Router, VueUse (200+ Utilities), Nuxt, Quasar, PrimeVue

📄 Single File Component (SFC): หัวใจของ Vue

ไฟล์ .vue คือสิ่งที่ทำให้ Vue พิเศษ — มันรวม HTML + JavaScript + CSS ไว้ในไฟล์เดียว แต่แยกส่วนกันชัดเจน

Anatomy แรกของไฟล์ .vue

<!-- ProductCard.vue -->
<script setup lang="ts">
// ═══════════════════════════════════════════
// 1. <script setup> — "สมอง" ของ Component
// ═══════════════════════════════════════════
// setup = Composition API + auto-import
// lang="ts" = ใช้ TypeScript (แนะนำเสมอ)
import { ref, computed } from 'vue'

// Props: ข้อมูลที่ Parent ส่งเข้ามา
interface Props {
  title: string
  price: number
  inStock?: boolean  // ? = optional (default: undefined)
}
const props = withDefaults(defineProps<Props>(), {
  inStock: true,
})

// Reactive State: ข้อมูลที่เปลี่ยนแปลงได้
const quantity = ref(1)

// Computed: ค่าที่คำนวณจาก state อื่น (auto cache!)
const totalPrice = computed(() => props.price * quantity.value)

// Methods
function increment() {
  quantity.value++
}

// Emits: ส่ง event กลับไปที่ Parent
const emit = defineEmits<{
  addToCart: [productTitle: string, qty: number]
}>()

function handleAddToCart() {
  emit('addToCart', props.title, quantity.value)
}
</script>

<!-- ═══════════════════════════════════════════ -->
<!-- 2. <template> — "ร่างกาย" ที่ User เห็น    -->
<!-- ═══════════════════════════════════════════ -->
<template>
  <div class="product-card">
    <h3>{{ title }}</h3>
    <p class="price">฿{{ totalPrice.toLocaleString() }}</p>

    <div v-if="inStock" class="controls">
      <button @click="quantity--" :disabled="quantity <= 1">−</button>
      <span>{{ quantity }}</span>
      <button @click="increment">+</button>
    </div>
    <p v-else class="out-of-stock">สินค้าหมด</p>

    <button class="btn-primary" @click="handleAddToCart">
      🛒 เพิ่มลงตะกร้า
    </button>
  </div>
</template>

<!-- ═══════════════════════════════════════════ -->
<!-- 3. <style scoped> — CSS เฉพาะ Component นี้  -->
<!-- ═══════════════════════════════════════════ -->
<style scoped>
/* scoped = CSS นี้จะไม่รั่วไปกระทบ Component อื่น! */
.product-card {
  border: 1px solid #e2e8f0;
  border-radius: 12px;
  padding: 20px;
}
.price { font-size: 24px; font-weight: bold; color: #10b981; }
.out-of-stock { color: #ef4444; }
.btn-primary {
  background: #3b82f6; color: white;
  border: none; border-radius: 8px;
  padding: 10px 20px; cursor: pointer;
}
</style>

ทำไม <script setup> ถึงดีกว่าแบบเดิม?

// ❌ แบบเก่า (Options API) — ยาว, ต้อง return ทุกอย่าง
export default {
  props: {...},
  data() { return { quantity: 1 } },
  computed: { totalPrice() { return this.price * this.quantity } },
  methods: { increment() { this.quantity++ } },
}

// ✅ แบบใหม่ (<script setup>) — สั้น, อ่านง่าย, TypeScript support ดีกว่า
// ทุกอย่างที่ประกาศใน <script setup> จะ "expose" ให้ template อัตโนมัติ!
// ไม่ต้อง return, ไม่ต้อง this, ไม่ต้อง methods: {}

กฎเหล็ก 2026: ใช้ <script setup> + Composition API เท่านั้น ยกเว้นดูแล Legacy Code

📝 Template Syntax: ภาษาของ Vue Template

Template ของ Vue คือ HTML ที่มี "พลังพิเศษ" เพิ่มเข้ามา เรียกว่า Directives (คำสั่ง v-)

1. Interpolation: แสดงข้อมูลใน Template

<template>
  <!-- Text Interpolation: ใช้ {{ }} -->
  <h1>{{ message }}</h1>

  <!-- Expression ได้ด้วย (ใส่ JavaScript expression ได้!) -->
  <p>{{ price * quantity }}</p>
  <p>{{ isLoggedIn ? 'สวัสดี' : 'กรุณาเข้าสู่ระบบ' }}</p>
  <p>{{ name.toUpperCase() }}</p>

  <!-- ⚠️ ห้ามใช้ statement (แค่ expression เท่านั้น!) -->
  <!-- ❌ {{ if (ok) { return yes } }}  → ใช้ ternary แทน -->
  <!-- ❌ {{ let x = 1 }}               → ประกาศตัวแปร ทำใน script -->
</template>

2. Directives: คำสั่ง v-* ที่ใช้บ่อยที่สุด

<template>
  <!-- ═══ v-if / v-else-if / v-else : แสดง/ซ่อน (จริงๆ คือ สร้าง/ทำลาย DOM) ═══ -->
  <div v-if="status === 'loading'">⏳ กำลังโหลด...</div>
  <div v-else-if="status === 'error'">❌ เกิดข้อผิดพลาด</div>
  <div v-else>✅ โหลดสำเร็จ!</div>

  <!-- ═══ v-show : แสดง/ซ่อน (ใช้ CSS display:none ไม่ทำลาย DOM) ═══ -->
  <!-- ใช้เมื่อ toggle บ่อยๆ (เช่น dropdown, tooltip) -->
  <div v-show="isDropdownOpen">เนื้อหา dropdown</div>

  <!-- ═══ v-for : วนลูป ═══ -->
  <ul>
    <li v-for="(item, index) in products" :key="item.id">
      {{ index + 1 }}. {{ item.name }} — ฿{{ item.price }}
    </li>
  </ul>
  <!-- ⚠️ :key สำคัญมาก! บอก Vue ว่า item ไหนคือใคร หาไม่ UI จะบั๊ก -->

  <!-- ═══ v-bind : ผูกค่า Attribute (ย่อ : ได้) ═══ -->
  <img :src="product.imageUrl" :alt="product.name">
  <button :disabled="!isValid" :class="{ active: isActive }">Submit</button>

  <!-- ═══ v-on : ผูก Event (ย่อ @ ได้) ═══ -->
  <button @click="handleClick">คลิกฉัน</button>
  <input @keyup.enter="submitForm">  <!-- .enter = Key Modifier -->
  <form @submit.prevent="onSubmit">   <!-- .prevent = preventDefault() -->

  <!-- ═══ v-model : Two-way Binding (ผูก input กับ data) ═══ -->
  <input v-model="searchQuery" placeholder="ค้นหา...">
  <!-- เทียบเท่ากับ: :value="searchQuery" @input="searchQuery = $event.target.value" -->

  <!-- v-model modifiers -->
  <input v-model.trim="name">         <!-- ตัด whitespace อัตโนมัติ -->
  <input v-model.number="age">        <!-- แปลงเป็น number อัตโนมัติ -->
  <input v-model.lazy="comment">      <!-- update ตอน blur แทน input -->
</template>

3. Class & Style Binding (ต้องใช้บ่อยมาก)

<template>
  <!-- Object Syntax: เปิด/ปิด class ตามเงื่อนไข -->
  <div :class="{ 
    'bg-green': isSuccess, 
    'bg-red': isError,
    'text-bold': true 
  }">...</div>

  <!-- Array Syntax: รวม class หลายๆ ค่า -->
  <div :class="[
    baseClass, 
    isActive ? 'active' : '', 
    errorClass
  ]">...</div>

  <!-- Dynamic Style -->
  <div :style="{
    color: activeColor,
    fontSize: fontSize + 'px',
    'background-color': bgColor   // kebab-case ต้องใส่ quote
  }">...</div>
</template>

⚡ Reactivity System: หัวใจสำคัญที่สุดของ Vue

ถ้าคุณเข้าใจ Reactivity = คุณเข้าใจ Vue ถ้าคุณไม่เข้าใจ Reactivity = คุณจะเขียน Vue ไม่ได้เลย

Analogy: Reactivity คืออะไร?

คิดว่า Reactivity เหมือน "Excel Spreadsheet" — เมื่อคุณเปลี่ยนค่าใน Cell A1 ทุก Cell ที่อ้างอิง A1 จะ คำนวณใหม่อัตโนมัติ ไม่ต้องกดปุ่มอะไร

Vue ทำเหมือนกัน: เมื่อ data เปลี่ยน → ทุก component ที่ใช้ data นั้นจะ update UI อัตโนมัติ

ref() — สำหรับค่าเดี่ยว (Primitive + Object)

<script setup lang="ts">
import { ref } from 'vue'

// ref() ทำให้ค่าเป็น "reactive" — Vue จะ track การเปลี่ยนแปลง
const count = ref(0)            // Ref<number>
const name = ref('Kidpech')     // Ref<string>
const isLoading = ref(false)    // Ref<boolean>

// ═══ กฎสำคัญ: ใน <script> ต้องใช้ .value ═══
function increment() {
  count.value++           // ✅ ต้องใช้ .value ใน script
  console.log(count.value)
}

// ═══ ใน <template> ไม่ต้องใช้ .value (Vue unwrap ให้อัตโนมัติ) ═══
</script>

<template>
  <p>{{ count }}</p>            <!-- ✅ ไม่ต้อง count.value -->
  <button @click="increment">+1</button>
</template>

reactive() — สำหรับ Object/Array

<script setup lang="ts">
import { reactive } from 'vue'

// reactive() ทำให้ Object ทั้งก้อนเป็น reactive
// ข้อดี: ไม่ต้องใช้ .value!
interface UserForm {
  name: string
  email: string
  age: number
}

const form = reactive<UserForm>({
  name: '',
  email: '',
  age: 0,
})

function updateName(newName: string) {
  form.name = newName   // ✅ ไม่ต้อง .value เลย!
}
</script>

<template>
  <input v-model="form.name" placeholder="ชื่อ">
  <input v-model="form.email" placeholder="อีเมล">
  <input v-model.number="form.age" placeholder="อายุ">
</template>

ref vs reactive — เลือกอะไรดี?

// ═══ RULE OF THUMB ═══
// 1. ใช้ ref() เป็น default สำหรับทุกอย่าง (ปลอดภัยที่สุด)
// 2. ใช้ reactive() เมื่อจัดกลุ่ม related state เช่น form

// ⚠️ ข้อควรระวัง reactive():
const state = reactive({ count: 0 })
// ❌ ห้าม destructure! จะสูญเสีย reactivity!
const { count } = state   // count ไม่ reactive แล้ว!
// ✅ ถ้าอยาก destructure ใช้ toRefs()
const { count } = toRefs(state)  // count เป็น Ref<number> ที่ reactive

// ❌ ห้าม reassign ทั้ง object!
state = { count: 1 }    // ❌ หลุด reactive!
Object.assign(state, { count: 1 })  // ✅ ทำแบบนี้แทน

🧮 Computed & Watchers: ดักจับการเปลี่ยนแปลง

Computed: ค่าคำนวณที่ Cache อัตโนมัติ

computed จะคำนวณค่าใหม่ เฉพาะเมื่อ dependency เปลี่ยน ไม่ได้คำนวณทุกครั้งที่ render

<script setup lang="ts">
import { ref, computed } from 'vue'

const products = ref([
  { name: 'iPhone', price: 35000, inStock: true },

watch() — ดัก "การเปลี่ยนแปลง" แล้วทำอะไรบางอย่าง

ใช้เมื่อต้อง "ทำ Side Effect" เช่น เรียก API, เขียน localStorage, ส่ง Analytics

<script setup lang="ts">
import { ref, watch, watchEffect } from 'vue'

const searchQuery = ref('')
const selectedCategory = ref('all')

// ═══ watch: ดักค่าเดียว ═══
watch(searchQuery, (newVal, oldVal) => {
  console.log(`Search changed: "${oldVal}" → "${newVal}"`)
  fetchSearchResults(newVal)  // เรียก API!
})

// ═══ watch หลายค่าพร้อมกัน ═══
watch([searchQuery, selectedCategory], ([newSearch, newCat], [oldSearch, oldCat]) => {
  fetchProducts({ search: newSearch, category: newCat })
})

// ═══ watch deep object ═══
const form = ref({ name: '', address: { city: '', zip: '' } })
watch(form, (newVal) => {
  console.log('Form changed!', newVal)
}, { deep: true })  // deep: true = track nested changes

// ═══ watch immediate (รันทันทีตอน mount) ═══
watch(searchQuery, (val) => {
  fetchResults(val)
}, { immediate: true })

// ═══ watchEffect: Auto-track dependencies ═══
// ไม่ต้องระบุว่า watch อะไร — Vue จะ detect เอง!
watchEffect(async () => {
  // Vue จะ "อ่าน" ว่าในนี้ใช้ ref อะไรบ้าง
  // แล้ว watch ทุกตัวที่ใช้อัตโนมัติ!
  const results = await fetchProducts({
    search: searchQuery.value,      // Vue detect: ต้อง watch searchQuery
    category: selectedCategory.value // Vue detect: ต้อง watch selectedCategory
  })
  products.value = results
})
</script>

เมื่อไหร่ใช้ computed vs watch vs watchEffect?

  • computed — เมื่อต้องการ ค่าที่คำนวณมา (derived value) เช่น filteredList, totalPrice
  • watch — เมื่อต้องทำ side effect ที่ควบคุมได้ (เรียก API, write to storage) และต้องการรู้ค่าเก่าด้วย
  • watchEffect — เมื่อต้องทำ side effect แต่ ไม่สน oldValue และอยากให้ auto-track ง่ายๆ

onUpdated(() => {
  console.log('5. onUpdated — re-render เสร็จแล้ว')
  // ⚠️ ระวัง: ถ้าแก้ reactive data ตรงนี้ จะ infinite loop!
})

// ═══ UNMOUNT PHASE (Component ถูกทำลาย) ═══
onBeforeUnmount(() => {
  console.log('6. onBeforeUnmount — กำลังจะถอด Component ออกจาก DOM')
})

onUnmounted(() => {
  console.log('7. onUnmounted — ถอดออกแล้ว ✅')
  // ✅ CLEANUP ที่นี่! ถ้าไม่ทำ = Memory Leak!
  clearInterval(intervalId)

  subscription.unsubscribe()
})

กฎเหล็ก Lifecycle

  • Fetch data → ใช้ onMounted หรือ watchEffect (ใน script setup ก็ได้ ถ้า SSR-compatible)
  • Init DOM-dependent library → ใช้ onMounted เท่านั้น!
  • Cleanup (timer, listener, subscription) → onUnmounted เสมอ!
  • ห้ามเปลี่ยน reactive data ใน onUpdated → infinite loop!

🧩 Composables: Logic ที่ใช้ซ้ำได้ (Vue's Superpower)

Composable คือ "ฟังก์ชั่น" ที่ใช้ Composition API ข้างใน ทำให้แยก Logic ออกมาใช้ซ้ำข้าม Component ได้

Composable ใน Vue = Custom Hook ใน React = Mixin แต่ดีกว่า 100 เท่า (ไม่มีปัญหา name collision, ไม่มี implicit dependency, TypeScript friendly)

ตัวอย่าง 1: useFetch — ดึงข้อมูลจาก API

// composables/useFetch.ts
import { ref, watchEffect, type Ref } from 'vue'

interface UseFetchReturn<T> {
  data: Ref<T | null>
  error: Ref<Error | null>
  isLoading: Ref<boolean>
  refresh: () => Promise<void>
}

export function useFetch<T>(url: string | Ref<string>): UseFetchReturn<T> {
  const data = ref<T | null>(null) as Ref<T | null>
  const error = ref<Error | null>(null)
  const isLoading = ref(false)

  async function fetchData() {
    isLoading.value = true
    error.value = null
    try {
      const urlValue = typeof url === 'string' ? url : url.value
      const response = await fetch(urlValue)
      if (!response.ok) throw new Error(`HTTP ${response.status}`)
      data.value = await response.json()
    } catch (e) {
      error.value = e as Error
    } finally {
      isLoading.value = false
    }
  }

  // ถ้า url เป็น ref → auto-refetch เมื่อ url เปลี่ยน!
  watchEffect(() => { fetchData() })

  return { data, error, isLoading, refresh: fetchData }
}

// composables/useDebounce.ts
import { ref, watch, type Ref } from 'vue'

export function useDebounce<T>(source: Ref<T>, delayMs = 300): Ref<T> {
  const debounced = ref(source.value) as Ref<T>
  let timer: ReturnType<typeof setTimeout>

  watch(source, (newVal) => {
    clearTimeout(timer)
    timer = setTimeout(() => {
      debounced.value = newVal
    }, delayMs)
  })

  return debounced
}

// composables/useLocalStorage.ts
import { ref, watch, type Ref } from 'vue'

export function useLocalStorage<T>(key: string, defaultValue: T): Ref<T> {
  // อ่านค่าเดิมจาก localStorage (ถ้ามี)
  const stored = localStorage.getItem(key)
  const data = ref<T>(stored ? JSON.parse(stored) : defaultValue) as Ref<T>

  // เมื่อค่าเปลี่ยน → เขียนลง localStorage อัตโนมัติ
  watch(data, (newVal) => {
    localStorage.setItem(key, JSON.stringify(newVal))
  }, { deep: true })

  return data
}

💉 Provide / Inject: ส่งข้อมูลข้าม Layers

  const user = ref<User | null>(null)
  const token = ref<string | null>(localStorage.getItem('token'))
  const isLoading = ref(false)

  // ═══ Getters ═══ (เหมือน computed)
  const isAuthenticated = computed(() => !!token.value)
  const isAdmin = computed(() => user.value?.role === 'admin')
  const displayName = computed(() => user.value?.name ?? 'Guest')

  // ═══ Actions ═══ (เหมือน function ธรรมดา — รองรับ async!)
  async function login(email: string, password: string) {
    isLoading.value = true
    try {
      const res = await apiClient.post('/auth/login', { email, password })
      token.value = res.data.token
      user.value = res.data.user
      localStorage.setItem('token', res.data.token)
    } catch (e) {
      throw e  // ให้ component จัดการ error
    } finally {
      isLoading.value = false
    }
  }

  function logout() {
    user.value = null
    token.value = null
    localStorage.removeItem('token')
  }

  async function fetchProfile() {
    if (!token.value) return
    const res = await apiClient.get('/auth/me')
    user.value = res.data
  }

  return { user, token, isLoading, isAuthenticated, isAdmin, displayName, login, logout, fetchProfile }
})

// directives/clickOutside.ts
import type { Directive } from 'vue'

export const vClickOutside: Directive<HTMLElement, () => void> = {
  mounted(el, binding) {
    // เก็บ handler ไว้ที่ element เพื่อจะได้ remove ได้
    (el as any).__clickOutsideHandler = (event: MouseEvent) => {
      // ถ้าคลิกนอก element → เรียก callback
      if (!el.contains(event.target as Node)) {
        binding.value()   // ฟังก์ชั่นที่ผ่านมาใน v-click-outside="fn"
      }
    }
    document.addEventListener('click', (el as any).__clickOutsideHandler)
  },
  unmounted(el) {
    document.removeEventListener('click', (el as any).__clickOutsideHandler)
  },
}
